import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

df = pd.read_csv('feature_selected_dataset (1).csv')

lags = ['return_lag_1', 'return_lag_2', 'return_lag_3', 'return_lag_5', 'return_lag_10']

for lag in lags:
  plt.figure(figsize=(6,4))
  sns.scatterplot(x=df[lag], y=df['target_next_return'], alpha=0.3)
  plt.title(f'{lag} vs target_next_return')
  plt.xlabel(lag)
  plt.ylabel('Next Day Return')
  plt.show()
